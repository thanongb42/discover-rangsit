8. ปัญหาที่เจอและวิธีแก้

curl -v https://discover.rangsitcity.go.th/ 2>&1 | tail -40

สาเหตุ: Nginx reload config ใหม่ไม่สำเร็จ เพราะ upstream ผิดชื่อ

sudo docker compose exec reverse-proxy nginx -T | grep -n "listen 443\|server_name discover"
[emerg] host not found in upstream "php-apache" in /etc/nginx/conf.d/discovery.conf:9
nginx: configuration file /etc/nginx/nginx.conf test failed

แก้จาก php-apache เป็น service จริง php-app

8.2 /city-map ได้ Apache 404

curl -s https://discover.rangsitcity.go.th/city-map | head -20

สาเหตุ: Nginx ส่ง /city-map ไป Apache ตรง ๆ ทำให้ไม่เข้า PHP router จึงต้อง proxy dynamic route ไป /discover_rangsit/public/index.php และส่ง path จริงผ่าน X-Original-URI

8.3 Regex location ใช้ proxy_pass แบบมี URI ไม่ได้

sudo docker compose exec reverse-proxy nginx -t
[emerg] "proxy_pass" cannot have URI part in location given by regular expression
nginx: configuration file /etc/nginx/nginx.conf test failed

แก้โดยเปลี่ยนจาก location ~ เป็น prefix/exact locations แยกราย path เช่น location /uploads/, location /images/, location = /robots.txt

9. Final Nginx config

ไฟล์สุดท้าย:

~/web-server/nginx/conf.d/discover.conf
server {
    listen 80;
    server_name discovery.rangsitcity.go.th discover.rangsitcity.go.th www.discover.rangsitcity.go.th;

    location /uploads/ {
        proxy_pass http://php-app:80/discover_rangsit/public/uploads/;
        proxy_set_header Host $host;
    }

    location /images/ {
        proxy_pass http://php-app:80/discover_rangsit/public/images/;
        proxy_set_header Host $host;
    }

    location /js/ {
        proxy_pass http://php-app:80/discover_rangsit/public/js/;
        proxy_set_header Host $host;
    }

    location /css/ {
        proxy_pass http://php-app:80/discover_rangsit/public/css/;
        proxy_set_header Host $host;
    }

    location /assets/ {
        proxy_pass http://php-app:80/discover_rangsit/public/assets/;
        proxy_set_header Host $host;
    }

    location /pr/ {
        proxy_pass http://php-app:80/discover_rangsit/public/pr/;
        proxy_set_header Host $host;
    }

    location = /presentation.html {
        proxy_pass http://php-app:80/discover_rangsit/public/presentation.html;
        proxy_set_header Host $host;
    }

    location = /robots.txt {
        proxy_pass http://php-app:80/discover_rangsit/public/robots.txt;
        proxy_set_header Host $host;
    }

    location = /favicon.ico {
        proxy_pass http://php-app:80/discover_rangsit/public/favicon.ico;
        proxy_set_header Host $host;
    }

    location / {
        proxy_pass http://php-app:80/discover_rangsit/public/index.php;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto http;
        proxy_set_header X-Original-URI $request_uri;
    }
}
